\subsection{Dart}

\subsection*{識別子}
Dartの識別子には以下のように大きく３種類に分けることが出来る。\par
\begin{itemize}
  \item \textbf{UpperCamelCase}：名前では、各単語の最初の文字が大文字になり、最初の文字も含まれる。
  \item \textbf{lowerCamelCase}：名前では、頭文字を覗き、各単語の最初の文字を大文字にする
  \item \textbf{lowercase\_with\_underscores}：頭字語であっても、小文字のみを使用し、単語は\_で区切る。
\end{itemize}
これを元に型やファイルなどの識別子を定める。

\subsubsection*{＜型にはUpperCamelCaseを用いる。＞}
クラス、列挙型、typedefs、型パラメータは、各単語の最初の文字を大文字にし（最初の単語を含む）、セパレータを使用しない。\par
\ \par

    \verb|class HttpRequest {...}|\par
    \verb|typedef Predicate<T> = bool Function(T value);|\par

\end{center}\par
\ \par

\subsubsection*{＜ライブラリ、パッケージ、ディレクトリ、ソースファイルにはlowercase\_with\_underscoresを使う＞}
ファイルシステムによっては大文字と小文字が区別されないため、
多くのプロジェクトでファイル名を全て小文字にする必要がある。
区切り文字を使用することで、
そのような形式でも名前を読むことができる。
アンダースコア（\_）をセパレータとして使用すると、
名前が有効なDart識別子であることが保証される。これは、
言語が後にシンボリックインポートをサポートする場合に便利である。\par
\ \par

    \ \par
    \verb|import 'file_system.dart';|\par
    \ \par
    \verb|import 'slider_menu.dart';|
  \end{tcolorbox}
\end{center}\par

\subsubsection*{＜改行＞}
\begin{itemize}
  \item 空行の場合はインデントを維持しない（スペースを削除する）\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|//引数がchildの1つ|\par
            \verb|Expand(child: Text('Hello')),|
          \end{tcolorbox}
        \end{center}\par
        \ \par
  \item 引数が2つ以上のWidgetは複数行で書く\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|// 引数がcolor, childの2つ|\par
            \verb|Container(|\par

            \ \ \verb|child: Text('Hello'), // Textは1行可|\par
            \verb|),|
          \end{tcolorbox}
        \end{center}\par
        \ \par

  \item Widgetの引数が1つでも、その子Widgetが複数行の場合は、改行。
        （1行でWidgetを書くのは、Widgetツリーの末端のみ）\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,

              title=good,
              width=14cm]

            \verb|//このWidget自身も複数行で書く|\par
            \verb|Container(|\par
            \ \ \verb|child: Text(|\par
            \ \ \verb|'Hello',|\par
            \ \ \verb|style: TextStyle(fontSize: 16.0),|\par
            \ \verb|),|\par
            \verb|),|
          \end{tcolorbox}
        \end{center}\par
        \ \par
        \clearpage
  \item 例外として、Edgelnsetsは複数の引数があっても１行で良い\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,

            \verb|),|
          \end{tcolorbox}
        \end{center}\par
\end{itemize}


\subsubsection*{＜括弧の位置＞}
\begin{itemize}
  \item 括弧の終了は、括弧の開始の後に改行しない場合は、同じ行で閉じる。\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|Container(child: Text('Hello')),|
          \end{tcolorbox}
        \end{center}\par
        \ \par

  \item 括弧の開始の後に改行した場合、
        中身の最終行の次の行で括弧を閉じる。
        尚、インデントはブロック開始時と同じにする。\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|Container(|\par
            \ \verb|color: Colors.blue,|\par
            \ \verb|child: Text('Hello'),|\par
            \verb|),|
          \end{tcolorbox}
        \end{center}\par
        \ \par

  \item 関数の呼び出しで「名前なし引数」で開始する場合、
        同じ行でWidgetの括弧を開始する。
        Widgetの閉じる括弧と関数呼び出しの閉じる括弧は同じ行。\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|someFunc(Container(|\par
            \ \verb|color: Colors.blue,|\par
            \ \verb|child: Text('Hello'),|\par
            \verb|)),|
          \end{tcolorbox}
        \end{center}\par
        \ \par
        \clearpage
  \item 関数の呼び出しで「名前あり引数」の場合は、
        Widgetと同様に記述。\par
        \ \par
        \begin{center}
          \begin{tcolorbox}[colframe=blue!7!green,
              colback=green!20!white,
              colbacktitle=green!93!blue!,
              coltitle=black, fonttitle=\bfseries,
              title=good,
              width=14cm]
            \verb|someFunc(|\par
            \ \verb|foo|\par
            \ \verb|barOption: true,|\par
            \ \verb|title: Text(...),|\par
            \verb|),|
          \end{tcolorbox}
        \end{center}\par
        \ \par
\end{itemize}
